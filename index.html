​const Highlighter = {
highlight: (code, lang) => {
if (!code) return '';
let html = code.replace(/&/g, '&').replace(//g, '>');
if (lang === 'html') {
html = html.replace(/()/g, '$1$2$3');
html = html.replace(/([a-zA-Z-]+)=/g, '$1=');
html = html.replace(/(".?")/g, '$1');
html = html.replace(/()/g, '$1');
} else if (lang === 'css') {
html = html.replace(/([a-zA-Z-]+)\s*:/g, '$1:');
html = html.replace(/:(.?);/g, ':$1;');
html = html.replace(/([.#@][a-zA-Z0-9_-]+)/g, '$1');
html = html.replace(//*[\s\S]?*//g, m => ${m});
} else if (lang === 'javascript' || lang === 'js' || lang === 'jsx' || lang === 'ts' || lang === 'tsx') {
const keywords = /\b(const|let|var|function|return|if|else|for|while|class|import|export|default|from|new|true|false|null|undefined|async|await|try|catch|window|document|useState|useEffect|useRef|useCallback|useMemo|React|ReactDOM)\b/g;
html = html.replace(keywords, '$1');
html = html.replace(/(".?"|'.?'|.*?)/g, '$1');
html = html.replace(/\b([0-9]+)\b/g, '$1');
html = html.replace(/([a-zA-Z0-9_]+)\s*(/g, '$1(');
html = html.replace(/()/g, '$1$2$3');
html = html.replace(/(//.|/*[\s\S]*?*/)/g, m => ${m});
}
return html;
}

const f = State.files.find(x => x.id === id);
const name = await Dialog.prompt("Rename File", "New filename:", f.name);
if (name && name.trim()) {
f.name = name.trim();
const ext = f.name.split('.').pop().toLowerCase();
f.language = (ext === 'js' || ext === 'jsx' || ext === 'ts' || ext === 'tsx') ? 'javascript' : ext;
Workspace.saveImmediate();
UI.render();
}
},
​deleteFile: async (e, id) => {
e.stopPropagation();
if (State.files.length <= 1) return Dialog.alert("Notice", "At least one file is required.");
if (await Dialog.confirm("Delete File", "Are you sure?")) {
State.files = State.files.filter(x => x.id !== id);
State.activeFileId = State.files[0].id;
Workspace.saveImmediate();
UI.render();
}
},
​deleteProj: async (id) => {
if (Object.keys(State.projects).length <= 1) return Dialog.alert("Notice", "At least one workspace must remain.");
if (await Dialog.confirm("Delete Workspace", "Delete this workspace?")) {
delete State.projects[id];
Workspace.saveImmediate();
Workspace.renderHomeGrid();
}
},
​copyContent: () => {
const f = State.files.find(x => x.id === State.activeFileId);
if (!f) return;
const ta = document.createElement('textarea');
ta.value = f.content;
document.body.appendChild(ta);
ta.select();
document.execCommand('copy');
document.body.removeChild(ta);
UI.showToast("Copied");
},
